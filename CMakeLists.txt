cmake_minimum_required(VERSION 3.22)
project(FamiliarNotepad VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Concurrent)
add_library(notepad-core src/storage.cpp src/storage.h src/recovery.cpp src/recovery.h)
target_link_libraries(notepad-core PUBLIC Qt6::Core)
target_include_directories(notepad-core PUBLIC src)
add_library(notepad-ui src/window.cpp src/window.h src/editor.cpp src/editor.h src/theme.cpp src/theme.h)
target_link_libraries(notepad-ui PUBLIC notepad-core Qt6::Widgets Qt6::Concurrent)
target_compile_definitions(notepad-ui PUBLIC NOTEPAD_VERSION="${PROJECT_VERSION}")
add_executable(familiar-notepad src/main.cpp packaging/resources.qrc)
target_link_libraries(familiar-notepad PRIVATE notepad-ui)
include(GNUInstallDirs)
install(TARGETS familiar-notepad RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.tcballard.FamiliarNotepad.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/io.github.tcballard.FamiliarNotepad.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(notepad-tests tests/tests.cpp)
  target_link_libraries(notepad-tests PRIVATE notepad-ui Qt6::Test)
  add_test(NAME notepad-tests COMMAND notepad-tests)
  set_tests_properties(notepad-tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(notepad-preview tests/preview.cpp)
  target_link_libraries(notepad-preview PRIVATE notepad-ui)
endif()
